Table customers {
  customer_id int [pk, increment]
  full_name varchar(100) [not null]
  email varchar(150) [unique, not null]
  phone varchar(20)
  address text
  created_at timestamp
}

Table categories {
  category_id int [pk, increment]
  name varchar(80) [unique, not null]
}

Table products {
  product_id int [pk, increment]
  category_id int [not null]
  name varchar(150) [not null]
  description text
  created_at timestamp
}

Table product_variants {
  variant_id int [pk, increment]
  product_id int [not null]
  sku varchar(40) [unique, not null]
  colour varchar(30)
  size varchar(20)
  lens_type varchar(30) [note: 'ANTI_BLUE_LIGHT or PHOTOCHROMIC']
  prescription_supported boolean
  price decimal(12,2) [not null]
  stock_quantity int [not null]
}

Table orders {
  order_id int [pk, increment]
  customer_id int [not null]
  order_date timestamp
  status varchar(20)
  total_amount decimal(12,2)
}

Table order_items {
  order_item_id int [pk, increment]
  order_id int [not null]
  variant_id int [not null]
  quantity int [not null]
  unit_price decimal(12,2) [not null]
}

Table payments {
  payment_id int [pk, increment]
  order_id int [unique, not null]
  amount decimal(12,2)
  method varchar(30)
  status varchar(20)
  paid_at timestamp
}

Table deliveries {
  delivery_id int [pk, increment]
  order_id int [unique, not null]
  address text
  status varchar(20)
  shipped_at timestamp
  delivered_at timestamp
}

Table reviews {
  review_id int [pk, increment]
  product_id int [not null]
  customer_id int [not null]
  rating int
  comment text
  created_at timestamp
}

Ref: products.category_id > categories.category_id
Ref: product_variants.product_id > products.product_id
Ref: orders.customer_id > customers.customer_id
Ref: order_items.order_id > orders.order_id
Ref: order_items.variant_id > product_variants.variant_id
Ref: payments.order_id - orders.order_id
Ref: deliveries.order_id - orders.order_id
Ref: reviews.product_id > products.product_id
Ref: reviews.customer_id > customers.customer_id
